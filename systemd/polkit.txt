polkit.addRule(function(action, subject) {
    // throw "Action ID = " + action.id;
    // polkit.log("Action ID: " + action.id);
    // polkit.log(JSON.stringify(action));
    if (
        (action.id == "org.freedesktop.systemd1.manage-units" || action.id == "org.freedesktop.systemd1.manage-unit-files")
        && subject.isInGroup("api")
    ) {
        return polkit.Result.YES;
    }
    // ( action.lookup("unit") == "logistics.service" || action.lookup("unit") == "api.service" )
});
